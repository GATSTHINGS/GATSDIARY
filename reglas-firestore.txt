rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function permitido(uid) {
      return request.auth != null
        && request.auth.uid == uid
        && request.auth.token.email_verified == true
        && request.auth.token.email in ['guimesot@gmail.com', 'raqgmraes@gmail.com'];
    }

    match /usuarios/{uid}/entradas/{id} {
      allow read, write: if permitido(uid);
    }

    // Lista de etiquetas
    match /usuarios/{uid}/ajustes/{id} {
      allow read, write: if permitido(uid);
    }
  }
}
